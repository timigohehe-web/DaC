\parttitle{一、选择题（1$\sim$12 题，每题 5 分，共 60 分）}

\begin{enumerate}

\item 设函数 $f(x)$ 在 $x=0$ 的某邻域内二阶可导，且 $f(0)=0$。若已知极限
\[ \lim_{x\to 0}\frac{f(x)-\sin x}{x^{2}}=2 \]
则下列结论正确的是（\quad）
\opts{(A) $x=0$ 是 $f(x)$ 的极大值点}{(B) $x=0$ 是 $f(x)$ 的极小值点}{(C) 点 $(0,0)$ 是曲线 $y=f(x)$ 的拐点}{(D) 曲线 $y=f(x)$ 在 $x=0$ 处有切线 $y=x$，但 $x=0$ 不是极值点}

\item 下列反常积分中，条件收敛的是（\quad）
\opts{(A) $\displaystyle\int_{1}^{+\infty}\frac{\sin x}{x^{2}}\,\mathrm{d}x$}{(B) $\displaystyle\int_{1}^{+\infty}\frac{\sin x}{\sqrt{x}}\,\mathrm{d}x$}{(C) $\displaystyle\int_{1}^{+\infty}\frac{\sin^{2}x}{x}\,\mathrm{d}x$}{(D) $\displaystyle\int_{0}^{1}\frac{\cos(1/x)}{\sqrt{x}}\,\mathrm{d}x$}

\item 关于无穷级数
\[ \sum_{n=1}^{\infty}\left[\frac{(-1)^{n}}{\sqrt{n}}+\frac{1}{n}\right] \quad\text{与}\quad \sum_{n=1}^{\infty}\ln\left(1+\frac{(-1)^{n}}{\sqrt{n}}\right) \]
的敛散性，下列判断正确的是（\quad）
\opts{(A) 两者均发散}{(B) 前者发散，后者条件收敛}{(C) 两者均条件收敛}{(D) 前者收敛，后者发散}

\item 设函数 $f(x,y)$ 在点 $(0,0)$ 的某邻域内有定义，且 $f(0,0)=0$。下列命题中正确的是（\quad）
\opts{(A) 若 $\displaystyle\lim_{(x,y)\to(0,0)}\frac{f(x,y)-ax-by}{\sqrt{x^{2}+y^{2}}}=0$，则偏导数 $f_x'(x,y)$ 在 $(0,0)$ 处必连续}{(B) 若 $f(x,y)$ 在 $(0,0)$ 处可微，则沿着任意非零向量 $\boldsymbol l$ 的方向导数 $\dfrac{\partial f}{\partial\boldsymbol l}(0,0)$ 均存在}{(C) 若偏导数 $f_x'(0,0)$ 与 $f_y'(0,0)$ 均存在，且沿任意直线 $y=kx$ 趋于 $(0,0)$ 时均有 $f(x,kx)=o(\sqrt{x^{2}+k^{2}x^{2}})$，则 $f(x,y)$ 在 $(0,0)$ 处必可微}{(D) 若 $f(x,y)$ 在 $(0,0)$ 处的两个偏导数 $f_x',f_y'$ 存在且有界，则 $f(x,y)$ 在 $(0,0)$ 处必连续}

\item 设常数 $a,b\in\mathbb{R}$，若微分方程 $y''+ay'+by=0$ 的所有解在整个实数轴 $(-\infty,+\infty)$ 上均有界，则参数 $a,b$ 满足的充要条件是（\quad）
\opts{(A) $a=0,\ b>0$}{(B) $a>0,\ b>0$}{(C) $a\ge 0,\ b>0$}{(D) $a=0,\ b\ge 0$}

\item 设三个平面方程构成的线性方程组为 $A\boldsymbol x=\boldsymbol b$，其中 $A$ 为 $3\times3$ 实矩阵。若矩阵的秩满足 $r(A)=2$ 且 $r(A,\boldsymbol b)=3$，且矩阵 $A$ 的任意两行向量均不成比例，则这三个平面在三维空间中的几何构型为（\quad）
\opts{(A) 两两相交于三条互相平行的直线，无公共点（空心三棱柱）}{(B) 存在两个互相平行的平面，第三个平面与它们分别相交}{(C) 三个平面交于同一条直线}{(D) 三个平面两两相交于同一条直线并重合}

\item 设 $A$ 为 $m\times n$ 实矩阵，$B$ 为 $n\times p$ 实矩阵。下列命题中错误的是（\quad）
\opts{(A) 齐次线性方程组 $A^{\mathrm{T}}A\boldsymbol x=\boldsymbol 0$ 与 $A\boldsymbol x=\boldsymbol 0$ 必为同解方程组}{(B) 若矩阵列满秩即 $r(A)=n$，则方阵 $A^{\mathrm{T}}A$ 必为正定矩阵}{(C) 齐次线性方程组 $AB\boldsymbol x=\boldsymbol 0$ 有非零解的充分必要条件是 $B\boldsymbol x=\boldsymbol 0$ 有非零解}{(D) 矩阵的秩恒满足 $r(A^{\mathrm{T}}A)=r(AA^{\mathrm{T}})=r(A)$}

\item 设 $A,B$ 均为 $n$ 阶实对称矩阵，下列命题中正确的是（\quad）
\opts{(A) 若 $A$ 为正定矩阵，则必存在可逆矩阵 $P$ 使得 $P^{-1}AP=E$}{(B) 若 $A,B$ 均正定，则矩阵乘积 $AB$ 必为正定矩阵}{(C) $A$ 正定的充分必要条件是矩阵 $A$ 与单位矩阵 $E$ 合同}{(D) 若 $A$ 正定，则其伴随矩阵 $A^{*}$ 不一定正定}

\item 设随机变量 $X$ 服从标准正态分布 $N(0,1)$，令随机变量 $Y=X^{2}$。记 $\rho_{XY}$ 为 $X$ 与 $Y$ 的相关系数，则（\quad）
\opts{(A) $X$ 与 $Y$ 相互独立，且 $\rho_{XY}=0$}{(B) $X$ 与 $Y$ 不独立，且 $\rho_{XY}=0$}{(C) $X$ 与 $Y$ 不独立，且 $\rho_{XY}\neq 0$}{(D) $X$ 与 $Y$ 相互独立，且 $\rho_{XY}\neq 0$}

\item 设元件 $A$ 与 $B$ 的寿命分别服从参数为 $p_1$ 与 $p_2$ 的几何分布（$0<p_1<p_2<1$）。库房中两类元件各占一半且外观无区别。随机抽取一件，记其寿命为 $T$。其离散失效率定义为 $\lambda(k)=P\{T=k\mid T\ge k\}$（$k=1,2,\dots$）。则关于失效率 $\lambda(k)$ 的性态，下列判断正确的是（\quad）
\opts{(A) $\lambda(k)$ 严格单调递增（老化加速）}{(B) $\lambda(k)$ 严格单调递减（老化减速，DHR）}{(C) $\lambda(k)$ 恒为常数（保持无记忆性）}{(D) 当 $k\to\infty$ 时，$\lambda(k)\to\dfrac{p_1+p_2}{2}$}

\item 设总体样本 $X_1,X_2,\dots,X_m$ 独立同分布于 $N(0,\sigma^{2})$，样本 $Y_1,Y_2,\dots,Y_n$ 独立同分布于 $N(0,2\sigma^{2})$，且两组样本相互独立。则下列构造出的统计量服从 $F(m,n)$ 分布的是（\quad）
\opts{(A) $\dfrac{n\sum_{i=1}^{m}X_i^{2}}{m\sum_{j=1}^{n}Y_j^{2}}$}{(B) $\dfrac{2n\sum_{i=1}^{m}X_i^{2}}{m\sum_{j=1}^{n}Y_j^{2}}$}{(C) $\dfrac{n\sum_{i=1}^{m}X_i^{2}}{2m\sum_{j=1}^{n}Y_j^{2}}$}{(D) $\dfrac{m\sum_{j=1}^{n}Y_j^{2}}{2n\sum_{i=1}^{m}X_i^{2}}$}

\item 设连续函数 $g(x)$ 在区间 $[0,1]$ 上严格单调递增，且积分满足 $\displaystyle\int_{0}^{1}g(x)\,\mathrm{d}x=0$。记定积分 $I=\displaystyle\int_{0}^{1}xg(x)\,\mathrm{d}x$，则下列结论恒成立的是（\quad）
\opts{(A) $I<0$}{(B) $I=0$}{(C) $I>0$}{(D) $I$ 的正负号取决于端点值 $g(0)$ 的符号}

\end{enumerate}

\parttitle{二、填空题（13$\sim$18 题，每题 5 分，共 30 分）}

\begin{enumerate}[resume]

\item 设函数 $y=y(x)$ 由隐式方程
\[ x=\int_{0}^{y-2x}\frac{1}{\sqrt{1+t^{4}}}\,\mathrm{d}t \]
确定，则极限 $\displaystyle\lim_{n\to\infty}n\left[y\left(\frac{1}{n}\right)-y(0)\right]=\blank$。

\item 微分方程 $yy''+(y')^{2}=0$ 满足初始条件 $y(0)=1$、$y'(0)=2$ 的特解为 $y(x)=\blank$。

\item 设空间四面体区域 $\Omega=\{(x,y,z)\mid x+y+z\le 1,\ x\ge 0,\ y\ge 0,\ z\ge 0\}$，则三重积分
\[ \iiint_{\Omega}(3x+2y+z)\,\mathrm{d}x\mathrm{d}y\mathrm{d}z=\blank \]

\item 设二元函数
\[ f(x,y)=\begin{cases}\dfrac{x^{3}y-xy^{3}}{x^{2}+y^{2}},&x^{2}+y^{2}\neq 0\\[2mm] 0,&x^{2}+y^{2}=0\end{cases} \]
则二阶混合偏导数 $f_{xy}''(0,0)=\blank$。

\item 设 3 阶实对称矩阵 $A$ 满足矩阵方程 $A^{2}=A$，且矩阵的秩 $r(A)=2$，则行列式 $|2A+E|=\blank$。

\item 设随机变量 $X,Y$ 相互独立且均服从标准正态分布 $N(0,1)$，则概率 $P\{X^{2}-Y^{2}>0\}=\blank$。

\end{enumerate}

\parttitle{三、解答题（19$\sim$27 题，共 60 分）}

\begin{enumerate}[resume]

\item（7 分）设函数 $f(x)$ 在闭区间 $[0,1]$ 上一阶连续可导，在开区间 $(0,1)$ 内二阶可导，且满足边界条件与定积分值：
\[ f(0)=0,\quad f(1)=1,\quad \int_{0}^{1}f(x)\,\mathrm{d}x=\frac{1}{3} \]
证明：在开区间 $(0,1)$ 内必存在一点 $\xi\in(0,1)$，使得 $f''(\xi)>0$。

\item（7 分）计算曲面积分
\[ I=\iint_{\Sigma}x^{3}\,\mathrm{d}y\mathrm{d}z+y^{3}\,\mathrm{d}z\mathrm{d}x+z\,\mathrm{d}x\mathrm{d}y \]
其中 $\Sigma$ 为抛物面 $z=1-x^{2}-y^{2}\ (z\ge 0)$ 的上侧。

\item（7 分）求幂级数 $\displaystyle\sum_{n=1}^{\infty}\frac{n}{n+1}x^{n}$ 的收敛域与和函数 $S(x)$。

\item（7 分）设矩阵
\[ A=\begin{pmatrix}1&0&1\\0&1&1\\1&1&2\end{pmatrix},\qquad B=\begin{pmatrix}1&1\\0&1\\1&2\end{pmatrix} \]
\begin{enumerate}
\item 证明 $A$ 的列向量组可由 $B$ 的列向量组线性表出，并求出分解式 $A=BC$（其中 $C$ 为 $2\times3$ 矩阵）；
\item 利用分解式 $A=BC$，求矩阵的高次幂 $A^{10}$。
\end{enumerate}

\item（7 分）设三维非零列向量 $\boldsymbol\alpha=(1,1,1)^{\mathrm{T}}$，定义 3 阶矩阵 $A=\boldsymbol\alpha\boldsymbol\alpha^{\mathrm{T}}-2E$。
\begin{enumerate}
\item 求矩阵 $A$ 的全部特征值及其对应的特征子空间维数；
\item 求一个可逆正交矩阵 $Q$，使得 $Q^{\mathrm{T}}AQ$ 为对角矩阵。
\end{enumerate}

\item（7 分）某工业系统由 $n$ 个独立工作的电子元件串联组成，每个元件的寿命 $X_i\ (i=1,2,\dots,n)$ 均服从参数为 $\lambda\ (\lambda>0)$ 的指数分布。当系统中第一个元件发生故障时系统立即停机。记系统工作寿命为 $T$。
\begin{enumerate}
\item 求系统寿命 $T$ 的概率密度函数 $f_T(t)$，并指出 $T$ 服从何种概率分布及其分布参数；
\item 计算系统的平均寿命 $E(T)$ 与方差 $D(T)$；
\item 若设计指标要求系统的平均寿命不得低于 $t_0\ (t_0>0)$，试求该系统所能串联容纳的最多元件个数 $n$。
\end{enumerate}

\item（6 分）设二维随机变量 $(X,Y)$ 的联合概率密度函数为
\[ f(x,y)=\begin{cases}c(x+y),&0<x<1,\ 0<y<1\\[1mm] 0,&\text{其他}\end{cases} \]
\begin{enumerate}
\item 求常数 $c$ 的值；
\item 求边缘概率密度函数 $f_X(x)$ 与 $f_Y(y)$；
\item 严格按照概率论关于随机变量独立的数学定义，判定 $X$ 与 $Y$ 是否相互独立，并给出严谨推导。
\end{enumerate}

\item（6 分）求微分方程 $yy''-2(y')^{2}=0$ 满足初始条件 $y(0)=1$、$y'(0)=1$ 的特解 $y(x)$。

\item（6 分）在三维空间直角坐标系中，求坐标原点 $(0,0,0)$ 到两条平面的交线
\[ \begin{cases}x+y+z=1\\ 2x-y+z=2\end{cases} \]
的最短距离。

\end{enumerate}
